cmake_minimum_required(VERSION 3.16)
project(wbrov_rviz_probe)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rviz_common REQUIRED)
find_package(rviz_rendering REQUIRED)
find_package(Qt5 REQUIRED COMPONENTS Widgets)
add_executable(rviz2_mono rviz2_mono.cpp)
target_compile_features(rviz2_mono PRIVATE cxx_std_17)
ament_target_dependencies(rviz2_mono rclcpp rviz_common rviz_rendering)
target_link_libraries(rviz2_mono Qt5::Widgets)
add_executable(test_startup_render_guard test_startup_render_guard.cpp)
target_compile_options(test_startup_render_guard PRIVATE -UNDEBUG)
enable_testing()
add_test(NAME startup_render_guard COMMAND test_startup_render_guard)
